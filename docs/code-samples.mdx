  <Tab title="TypeScript / Node.js">
```typescript
import crypto from 'crypto';

interface SoftpayPayload {
  amount: string;
  currency: string;
  country: string;
  network: string;
  customer: {
    first_name: string;
    last_name: string;
    email: string;
    phone: string;
  };
  description: string;
}

async function createPayment() {
  const apiKey = process.env.PICHFLOW_API_KEY || 'sk_test_...';
  const idempotencyKey = crypto.randomUUID();

  const payload: SoftpayPayload = {
    amount: "15000.00",
    currency: "XOF",
    country: "BJ",
    network: "mtn_bj",
    customer: {
      first_name: "Paul",
      last_name: "Dossou",
      email: "paul@example.com",
      phone: "+22997001234"
    },
    description: "Licence Annuelle PichFlow"
  };

  const response = await fetch('https://api.pichflow.com/v1/payments/softpay', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${apiKey}`,
      'Content-Type': 'application/json',
      'Idempotency-Key': idempotencyKey
    },
    body: JSON.stringify(payload)
  });

  const data = await response.json();
  console.log('Réponse de paiement :', data);
}

createPayment();
```

  <Tab title="Python">
```python
import os
import uuid
import requests

API_KEY = os.getenv("PICHFLOW_API_KEY", "sk_test_...")
BASE_URL = "https://api.pichflow.com/v1"

payload = {
    "amount": "15000.00",
    "currency": "XOF",
    "country": "BJ",
    "network": "mtn_bj",
    "customer": {
        "first_name": "Paul",
        "last_name": "Dossou",
        "email": "paul@example.com",
        "phone": "+22997001234"
    },
    "description": "Licence Annuelle PichFlow"
}

headers = {
    "Authorization": f"Bearer {API_KEY}",
    "Content-Type": "application/json",
    "Idempotency-Key": str(uuid.uuid4())
}

response = requests.post(f"{BASE_URL}/payments/softpay", json=payload, headers=headers)
print("Statut:", response.status_code)
print("Réponse:", response.json())
```

  <Tab title="PHP">
```php
<?php

$apiKey = getenv('PICHFLOW_API_KEY') ?: 'sk_test_...';
$idempotencyKey = bin2hex(random_bytes(16));

$payload = [
    'amount' => '15000.00',
    'currency' => 'XOF',
    'country' => 'BJ',
    'network' => 'mtn_bj',
    'customer' => [
        'first_name' => 'Paul',
        'last_name' => 'Dossou',
        'email' => 'paul@example.com',
        'phone' => '+22997001234'
    ],
    'description' => 'Licence Annuelle PichFlow'
];

$ch = curl_init('https://api.pichflow.com/v1/payments/softpay');
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($payload));
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    'Authorization: Bearer ' . $apiKey,
    'Content-Type: application/json',
    'Idempotency-Key: ' . $idempotencyKey
]);

$response = curl_exec($ch);
$httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
curl_close($ch);

echo "Code HTTP: " . $httpCode . "\n";
echo "Réponse: " . $response . "\n";
?>
```
